# Copy it into its own folder (mkdir my-eval && cp howto/promptfooconfig.yaml my-eval/), change it, run it.
# Target: the lab's Helpdesk on this computer only. Start it first: python3 ../helpdesk.py serve &
description: "My first eval: TSG Helpdesk"
prompts:
  - "{{message}}"
providers:
  - id: http
    config:
      url: http://127.0.0.1:8017/api/chat
      method: POST
      headers: {Content-Type: application/json}
      body: {user: bob, message: "{{prompt}}"}
      # The reply is JSON. Turn it into text a person can read: the answer, then one line per tool call.
      transformResponse: 'json.answer + "\n" + json.tool_calls.map(c => "TOOL " + c.tool + " " + (c.args.to || c.args.user || c.args.id) + " " + c.decision).join("\n")'
