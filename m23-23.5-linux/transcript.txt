"""defense.py: my five checks. Each reads one recorded event and returns a verdict. Behaviour, not wording."""
from __future__ import annotations

import re
from datetime import datetime
from urllib.parse import urlsplit

OVERRIDE = re.compile(r"(?i)\b(ignore|disregard|forget)\b[^.]{0,20}\b(previous|prior|earlier|above)\b"
                      r"[^.]{0,10}\b(instructions|directions|rules)\b|\byou\s+are\s+now\b|^\s*system:")
HIDDEN = re.compile(r"<!--.*?-->", re.S)
ZERO_WIDTH = re.compile(r"[\u200b\u200c\u200d\u2060\ufeff]")
URL = re.compile(r"https?://[^\s\"'<>)]+")


def input_check(event: dict, config: dict) -> str | None:
    text = event.get("text", "")
    if HIDDEN.search(text):
        return "text hidden in an HTML comment"
    if ZERO_WIDTH.search(text):
        return "zero-width characters"
    if OVERRIDE.search(ZERO_WIDTH.sub("", text)):
        return "override phrasing aimed at instructions"
    return None

def allowlist(tool_call: dict, config: dict) -> bool:
    tool, args = tool_call.get("tool"), tool_call.get("args", {})
    if tool not in config["allowed_tools"]:
        return False
    if tool == "send_email":
        return args.get("to", "").rsplit("@", 1)[-1].lower() in config["allowed_email_domains"]
    if tool == "fetch_url":
        return (urlsplit(args.get("url", "")).hostname or "") in config["allowed_fetch_hosts"]
    return True


def output_check(event: dict, config: dict) -> str | None:
    for url in URL.findall(event.get("text", "")):
        parts = urlsplit(url)
        if parts.hostname not in config["allowed_fetch_hosts"] and parts.query:
            return f"data in a link to an outside host: {parts.hostname}"
    return None

def rate_anomaly(login_events: list[dict], config: dict) -> set[str]:
    budget, fails = config["rate"]["login_fail_per_min"], {}
    for e in login_events:
        if e.get("result") == "fail":
            fails.setdefault(e["actor"], []).append(datetime.fromisoformat(e["ts"].replace("Z", "+00:00")))
    flagged = set()
    for actor, times in fails.items():
        times.sort()
        for i, start in enumerate(times):
            if sum(0 <= (t - start).total_seconds() <= 60 for t in times[i:]) > budget:
                flagged.add(actor)
    return flagged


def egress_signature(egress: dict, config: dict) -> str | None:
    dest, host = egress.get("dest", "").lower(), egress.get("host", "")
    is_model_api = any(dest == d or dest.endswith("." + d) for d in config["model_api_domains"])
    if is_model_api and host not in config["model_callers"]:
        return f"model API call from {host}, which is not a model caller"
    return None
